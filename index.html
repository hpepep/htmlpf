async function calculatePortfolio() { const jsonInputEl = document.getElementById('jsonInput'); const tableBody = document.getElementById('tableBody'); const totalFundsEl = document.getElementById('totalFunds'); if (!jsonInputEl || !tableBody) return; const jsonText = jsonInputEl.value; let rawData; try { rawData = JSON.parse(jsonText); } catch (e) { alert("Invalid JSON format inside code block."); return; } const fundsBlock = rawData.mutual_funds || {}; const amfiCodes = Object.keys(fundsBlock); if (amfiCodes.length === 0) { tableBody.innerHTML = `No mutual funds identified.`; document.getElementById('mfCurrentVal').innerText = "₹0.00"; globalMFCurrent = 0; globalMFFuture = 0; updateGlobalOverview(); return; } tableBody.innerHTML = `Querying live AMFI pricing index...`; let grandTotal = 0; let successCount = 0; localizedCacheRows = []; const fetchPromises = amfiCodes.map(async (amfi) => { const item = fundsBlock[amfi]; if (!item || typeof item.units === 'undefined') return; const units = parseFloat(item.units) || 0; const manualName = item.name || item.manualName; try { // Switch to the optimized /latest endpoint const response = await fetch(`https://api.mfapi.in/mf/${amfi}/latest`); const json = await response.json(); let schemeName = manualName; let currentNav = 0; if (json && json.data && json.data.length > 0) { currentNav = parseFloat(json.data[0].nav) || 0; if (!schemeName && json.meta && json.meta.scheme_name) { schemeName = json.meta.scheme_name; } } if (!schemeName) { schemeName = `Unindexed Scheme [AMFI: ${amfi}]`; } if (currentNav > 0) { const currentValue = units * currentNav; return { name: schemeName, code: parseInt(amfi), units: units, nav: currentNav, value: currentValue, status: 'success' }; } else { throw new Error("Missing NAV"); } } catch (e) { return { name: manualName || `Lookup Exception: Scheme metadata error [AMFI: ${amfi}]`, code: parseInt(amfi), units: units, nav: 0, value: 0, status: 'error' }; } }); const results = await Promise.all(fetchPromises); localizedCacheRows = results.filter(row => row !== undefined); localizedCacheRows.forEach(row => { if (row.status === 'success') { grandTotal += row.value; successCount++; } }); globalMFCurrent = grandTotal; const rYear = parseInt(document.getElementById('retirementYear').value) || 2043; const yearsLeft = Math.max(0, rYear - 2026); globalMFFuture = grandTotal * Math.pow(1.10, yearsLeft); document.getElementById('mfCurrentVal').innerText = `₹${grandTotal.toLocaleString('en-IN', { minimumFractionDigits: 2 })}`; if (totalFundsEl) totalFundsEl.innerText = successCount; const lastUpdatedEl = document.getElementById('lastUpdated'); if (lastUpdatedEl) { const timeMarker = new Date(); lastUpdatedEl.innerText = `Last updated: ${timeMarker.toLocaleTimeString()}`; } applySort(); updateGlobalOverview(); }
